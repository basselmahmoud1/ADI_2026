Function twiddle_Wx 

function Wx_ROM = twiddle_Wx(N) %#codegen 

Wx_ROM = complex(zeros(1, N/2));
for indx = 1:N/2
    k = indx - 1;
    Wx_ROM(indx) = exp(-1i * 2 * pi * k / N);
end

end

Function sdf_stage_1 

function data_out = sdf_stage_1(in_elem, clk)%#codegen
    DELAY_SIZE = 4;
    READ = false;
    WRITE = true;
    
    persistent W8_ROM;
    if isempty(W8_ROM)
        W8_ROM = twiddle_Wx(8);
    end

    % Use 0-indexed counter for easier logic (0 to 7)
    counter = mod(clk, 8); 

    % Always read from the delay line first
    delay_out = delay_x(0, DELAY_SIZE, READ);

    if counter < 4 
        % Cycles 0, 1, 2, 3: Orange Path "first data (out is garbage)"
        % 1. Multiply the delayed SUB by twiddle factor and output it
        % 2. Store the incoming data into the delay line
        twiddle_idx = counter + 1;
        W = W8_ROM(twiddle_idx);
        
        data_out = delay_out * W;
        delay_x(in_elem, DELAY_SIZE, WRITE);
    else 
        % Cycles 4, 5, 6, 7: Blue Path "now the delay stores the " 
        % 1. Compute butterfly with delayed data (A) and incoming data (B)
        % 2. Output the SUM
        % 3. Store the SUB into the delay line
        [SUM, SUB] = butterfly_R2(delay_out, in_elem);
        
        data_out = SUM;
        delay_x(SUB, DELAY_SIZE, WRITE);
    end 
end
